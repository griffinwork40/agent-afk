# Hash-pinned harness dependency for the nightly Harbor eval.
#
# Single source of truth for the harbor wheel hash.  CI and nightly workflows
# read this value with:
#   grep '^# HARBOR_SHA256=' bench/harbor/requirements-harness.txt | cut -d= -f2
# Regenerate when bumping harbor_version in nightly.json:
#   pip download 'harbor==<ver>' --no-deps --dest /tmp/d && pip hash /tmp/d/*.whl
#
# HARBOR_SHA256=8747400dbb2a5e2298e1338e17e88eba38433c0433fd700f34d1a9021bba5c37
#
# pip's --require-hashes mode forces ALL transitive deps to carry hashes too.
# Instead we download the wheel once, verify its hash against the line above,
# then install directly from the verified local copy (--find-links + --no-index)
# so pip never re-fetches from PyPI.
harbor==0.23.0
